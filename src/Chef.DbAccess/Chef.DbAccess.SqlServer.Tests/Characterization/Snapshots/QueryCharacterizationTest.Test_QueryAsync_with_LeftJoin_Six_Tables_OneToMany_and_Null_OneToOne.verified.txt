SELECT [use_1].[Id], [use_1].[Name], [use_2].[Id], [use_2].[Name], [use_3].[Id], [use_3].[Name], [dep_4].[Id] AS [DepId], [dep_4].[Name]
FROM [Member] [use_1] WITH (NOLOCK)
LEFT JOIN [Member] [use_2] WITH (NOLOCK) ON [use_1].[Id] = [use_2].[ManagerId]
LEFT JOIN [Member] [use_3] WITH (NOLOCK) ON [use_2].[Id] = [use_3].[ManagerId]
LEFT JOIN [Department] [dep_4] WITH (NOLOCK) ON [use_3].[DepartmentId] = [dep_4].[Id]
WHERE [use_1].[Id] = {=Id_0} OR [use_1].[Id] = {=Id_1};
-- @Id_0 = 1 (Int32)
-- @Id_1 = 2 (Int32)
-- splitOn: Id,Id,DepId
