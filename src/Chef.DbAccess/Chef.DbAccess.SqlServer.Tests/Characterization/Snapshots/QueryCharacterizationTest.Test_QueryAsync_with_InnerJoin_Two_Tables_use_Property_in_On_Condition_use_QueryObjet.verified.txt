SELECT [use_1].[Id], [use_1].[Name], [dep_2].[Id] AS [DepId], [dep_2].[Name]
FROM [Member] [use_1] WITH (NOLOCK)
INNER JOIN [Department] [dep_2] WITH (NOLOCK) ON ([use_1].[DepartmentId] = [dep_2].[Id]) AND (([use_1].[DepartmentId] = {=DepartmentId_0}) OR ([use_1].[DepartmentId] = {=DepartmentId_1}))
WHERE [use_1].[Id] = {=Id_0} OR [use_1].[Id] = {=Id_1};
