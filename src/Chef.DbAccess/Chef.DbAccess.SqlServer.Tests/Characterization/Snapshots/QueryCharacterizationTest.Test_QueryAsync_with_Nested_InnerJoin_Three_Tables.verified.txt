SELECT [use_1].[Id], [use_1].[Name], [use_2].[Id], [use_2].[Name], [dep_3].[Id] AS [DepId], [dep_3].[Name]
FROM [Member] [use_1] WITH (NOLOCK)
INNER JOIN [Member] [use_2] WITH (NOLOCK) ON [use_1].[ManagerId] = [use_2].[Id]
INNER JOIN [Department] [dep_3] WITH (NOLOCK) ON [use_2].[DepartmentId] = [dep_3].[Id]
WHERE [use_1].[Id] = {=Id_0} OR [use_1].[Id] = {=Id_1};
-- @Id_0 = 1 (Int32)
-- @Id_1 = 3 (Int32)
-- splitOn: Id,DepId
