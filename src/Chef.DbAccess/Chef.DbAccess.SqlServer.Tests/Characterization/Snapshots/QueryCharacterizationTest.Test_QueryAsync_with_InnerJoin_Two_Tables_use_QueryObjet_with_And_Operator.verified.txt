SELECT [use_1].[Id], [use_1].[Name], [dep_2].[Id] AS [DepId], [dep_2].[Name]
FROM [Member] [use_1] WITH (NOLOCK)
INNER JOIN [Department] [dep_2] WITH (NOLOCK) ON ([use_1].[DepartmentId] = [dep_2].[Id]) AND (([dep_2].[Id] & {=DepId_0}) > {=DepId_1});
-- @DepId_0 = 1 (Int32)
-- @DepId_1 = 0 (Int32)
-- splitOn: DepId
