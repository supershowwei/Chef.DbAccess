(
    [Name] NVARCHAR(MAX), [ClubID] INT
)

SET [Name] = tvp.[Name]
FROM [Club] t
INNER JOIN @TableVariable tvp
    ON (t.[ClubID] >= tvp.[ClubID]) AND (t.[ClubID] <= tvp.[ClubID])

INSERT INTO [Club]([Name], [ClubID])
    SELECT tvp.[Name], tvp.[ClubID]
    FROM @TableVariable tvp
    WHERE NOT EXISTS (SELECT
                1
            FROM [Club] t WITH (NOLOCK)
            WHERE (t.[ClubID] >= tvp.[ClubID]) AND (t.[ClubID] <= tvp.[ClubID]));

CREATE TYPE UDT_Club_Type2 AS TABLE
(
    [Name] NVARCHAR(MAX), [ClubID] INT
)

-- BEGIN TRANSACTION

UPDATE [Club]
SET [Name] = tvp.[Name]
FROM [Club] t
INNER JOIN @TableVariable tvp
    ON t.[ClubID] = tvp.[ClubID]

INSERT INTO [Club]([Name], [ClubID])
    SELECT tvp.[Name], tvp.[ClubID]
    FROM @TableVariable tvp
    WHERE NOT EXISTS (SELECT
                1
            FROM [Club] t WITH (NOLOCK)
            WHERE t.[ClubID] = tvp.[ClubID]);
