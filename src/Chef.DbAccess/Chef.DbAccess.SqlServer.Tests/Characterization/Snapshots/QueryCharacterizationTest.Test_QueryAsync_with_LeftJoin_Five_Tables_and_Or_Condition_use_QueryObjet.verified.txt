SELECT [use_1].[Id], [use_1].[Name], [dep_2].[Id] AS [DepId], [dep_2].[Name], [use_3].[Id], [use_3].[Name], [dep_4].[Id] AS [DepId], [dep_4].[Name], [use_5].[Id], [use_5].[Name]
FROM [Member] [use_1] WITH (NOLOCK)
LEFT JOIN [Department] [dep_2] WITH (NOLOCK) ON [use_1].[DepartmentId] = [dep_2].[Id]
LEFT JOIN [Member] [use_3] WITH (NOLOCK) ON [use_1].[Id] = [use_3].[Id]
LEFT JOIN [Department] [dep_4] WITH (NOLOCK) ON [use_3].[DepartmentId] = [dep_4].[Id]
INNER JOIN [Member] [use_5] WITH (NOLOCK) ON [use_3].[ManagerId] = [use_5].[Id]
WHERE ([use_1].[Id] = {=Id_0}) OR ([use_1].[Id] = {=Id_1});
-- @Id_0 = 1 (Int32)
-- @Id_1 = 4 (Int32)
